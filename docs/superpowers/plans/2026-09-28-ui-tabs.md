# `@libs/ui/tabs` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the `@libs/ui/tabs` secondary entry point wrapping `@angular/aria/tabs` with compound directives, Tailwind CSS v4 design tokens, CVA variants (`bordered`, `lift`, `pill`), horizontal/vertical orientations, lazy content templates, comprehensive tests, and an interactive docs playground.

**Architecture:** Build compound directives using Angular `hostDirectives` on `@angular/aria/tabs` (`Tabs`, `TabList`, `Tab`, `TabPanel`, `TabContent`). `UiTabsDirective` creates and provides a reactive context (`UI_TABS_CONTEXT`) tracking `variant`, `size`, `orientation`, and `color`. `UiTabListDirective` formats the container and orientation, and `UiTabDirective` applies reactive CVA styling with active ARIA state attributes. `UiTabContentDirective` provides lazy-loaded panel DOM instantiations.

**Tech Stack:** Angular 19+ (signals, `hostDirectives`, `input()`, `computed()`), `@angular/aria/tabs`, Tailwind CSS v4 `@utility`, CVA (`@libs/ui/core`), Vitest.

**Spec:** [`docs/superpowers/specs/2026-09-28-ui-tabs-design.md`](file:///home/tiendat/Code/angular-starter/docs/superpowers/specs/2026-09-28-ui-tabs-design.md)

## Global Constraints

- Standalone directives only; OnPush change detection strategy.
- No `zone-testing.js` `fakeAsync` or `tick` in specs; use Vitest fake timers (`vi.useFakeTimers()` / `vi.advanceTimersByTimeAsync()`) or native async/await.
- All styles implemented in Tailwind CSS v4 `@utility` classes in `libs/ui/styles/components/tabs.css` using semantic `--color-*` and `--radius-*` tokens.
- No direct coupling to app; pure secondary entry point `@libs/ui/tabs` exported through root `libs/ui/src/public-api.ts`.
- Package build must succeed via `npx ng build ui` and pass all tests via `npx ng test ui --watch=false`.

## Review Focus

1. **Disabled tabs keyboard navigation:** When a tab has `[disabled]="true"`, arrow key navigation must skip it and jump directly to the next enabled tab.
2. **Vertical orientation arrow navigation:** In `tabs-vertical`, ArrowDown/ArrowUp must navigate between tabs and `aria-orientation="vertical"` must be set on the tablist.
3. **Lazy tab content instantiation:** Elements inside `<ng-template uiTabContent>` must not exist in DOM until the parent `[uiTabPanel]` becomes active.
4. **Context reactivity:** Changing `[uiTabsVariant]`, `[uiTabsSize]`, or `[uiTabsColor]` dynamically on `[uiTabs]` must reactively update styling classes on all child `[uiTab]` buttons.
5. **WAI-ARIA roles & linkage:** `role="tablist"`, `role="tab"`, `role="tabpanel"`, and `aria-controls` / `aria-labelledby` IDs must be automatically linked and verifiable in DOM.

import { Component, signal } from '@angular/core';
import { ComponentFixture, TestBed } from '@angular/core/testing';
import { describe, expect, it, beforeEach } from 'vitest';
import {
  UiTabsDirective,
  UiTabListDirective,
  UiTabDirective,
  UiTabPanelDirective,
  UiTabContentDirective,
} from './public-api';
import { UiTabsOrientation, UiTabsVariant } from './tabs.types';

@Component({
  standalone: true,

    UiTabListDirective,

      uiTabs
      [uiTabsVariant]="variant()"

        [selectedTab]="selectedTab()"

      <div uiTabPanel value="tab2" id="panel2">
        <ng-template uiTabContent>
          <span class="panel2-content">Content 2</span>
        </ng-template>
      </div>

      <div uiTabPanel value="tab3" id="panel3">
        <ng-template uiTabContent>
          <span class="panel3-content">Content 3</span>
        </ng-template>
      </div>
    </div>
  `,
})
class TestHostComponent {
  readonly variant = signal<UiTabsVariant>('bordered');
  readonly orientation = signal<UiTabsOrientation>('horizontal');
  readonly selectedTab = signal<string>('tab1');
  readonly tab2Disabled = signal<boolean>(false);
}

describe('UiTabs Full Integration', () => {
  let fixture: ComponentFixture<TestHostComponent>;

  beforeEach(() => {
    TestBed.configureTestingModule({
      imports: [TestHostComponent],
    });
    fixture = TestBed.createComponent(TestHostComponent);
    fixture.detectChanges();
  });

  it('renders initial active tab with aria-selected="true" and associated panel content', () => {
    const tab1: HTMLElement = fixture.nativeElement.querySelector('#tab1-btn');
    const tab2: HTMLElement = fixture.nativeElement.querySelector('#tab2-btn');
    expect(tab1.getAttribute('aria-selected')).toBe('true');
    expect(tab2.getAttribute('aria-selected')).toBe('false');

    // Panel 1 lazy content is rendered
    expect(fixture.nativeElement.querySelector('.panel1-content')).not.toBeNull();
    // Inactive panels have not rendered their lazy template content
    expect(fixture.nativeElement.querySelector('.panel2-content')).toBeNull();
    expect(fixture.nativeElement.querySelector('.panel3-content')).toBeNull();
  });

  it('switches tabs on click and lazy-loads corresponding panel content', () => {
    const tab3: HTMLElement = fixture.nativeElement.querySelector('#tab3-btn');
    tab3.click();
    fixture.detectChanges();

    expect(fixture.componentInstance.selectedTab()).toBe('tab3');
    expect(tab3.getAttribute('aria-selected')).toBe('true');
    expect(fixture.nativeElement.querySelector('.panel3-content')).not.toBeNull();
  });

  it('navigates with ArrowRight and ArrowLeft in horizontal mode', () => {
    const tab1: HTMLElement = fixture.nativeElement.querySelector('#tab1-btn');
    tab1.focus();

    tab1.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowRight', bubbles: true }));
    fixture.detectChanges();

    expect(fixture.componentInstance.selectedTab()).toBe('tab2');
    const tab2: HTMLElement = fixture.nativeElement.querySelector('#tab2-btn');
    expect(tab2.getAttribute('aria-selected')).toBe('true');
  });

  it('skips disabled tabs during keyboard navigation', () => {
    fixture.componentInstance.tab2Disabled.set(true);
    fixture.detectChanges();

    const tab1: HTMLElement = fixture.nativeElement.querySelector('#tab1-btn');
    tab1.focus();

    tab1.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowRight', bubbles: true }));
    fixture.detectChanges();

    // Skipped tab2 and jumped to tab3!
    expect(fixture.componentInstance.selectedTab()).toBe('tab3');
    const tab3: HTMLElement = fixture.nativeElement.querySelector('#tab3-btn');
    expect(tab3.getAttribute('aria-selected')).toBe('true');
  });

  it('navigates with ArrowDown and ArrowUp in vertical mode', () => {
    fixture.componentInstance.orientation.set('vertical');
    fixture.detectChanges();

    const tab1: HTMLElement = fixture.nativeElement.querySelector('#tab1-btn');
    tab1.focus();

    tab1.dispatchEvent(new KeyboardEvent('keydown', { key: 'ArrowDown', bubbles: true }));
    fixture.detectChanges();

    expect(fixture.componentInstance.selectedTab()).toBe('tab2');
  });

  it('jumps to first and last tabs with Home and End keys', () => {
    const tab1: HTMLElement = fixture.nativeElement.querySelector('#tab1-btn');
    tab1.focus();

    tab1.dispatchEvent(new KeyboardEvent('keydown', { key: 'End', bubbles: true }));
    fixture.detectChanges();
    expect(fixture.componentInstance.selectedTab()).toBe('tab3');

    tab1.dispatchEvent(new KeyboardEvent('keydown', { key: 'Home', bubbles: true }));
    fixture.detectChanges();
    expect(fixture.componentInstance.selectedTab()).toBe('tab1');
  });
});
